--- name: Legend Group Click Behavior language: plotly_js suite: legends order: 11.5 sitemap: false arrangement: horizontal markdown_content: | A click on a grouped legend item toggles the whole group. `legend.groupclick` changes that. Set it to `toggleitem`, and a click toggles only the item under the cursor. *New in 4.1* `legend.groupdoubleclick` makes the same choice for a double-click. It takes the same two values, and it defaults to the value of `groupclick`. A double-click also isolates what it acts on, because `legend.itemdoubleclick` defaults to `toggleothers`. In the example below, a click toggles one trace. A double-click shows one whole group and hides the other. --- var trace1 = { x: [1, 2, 3], y: [2, 1, 3], legendgroup: 'group', marker: {color: 'rgb(164, 194, 244)'}, mode: 'markers', name: 'first legend group', type: 'scatter' }; var trace2 = { x: [1, 2, 3], y: [2, 2, 2], legendgroup: 'group', line: {color: 'rgb(164, 194, 244)'}, mode: 'lines', name: 'first legend group - average', type: 'scatter' }; var trace3 = { x: [1, 2, 3], y: [4, 9, 2], legendgroup: 'group2', marker: {color: 'rgb(142, 124, 195)'}, mode: 'markers', name: 'second legend group', type: 'scatter' }; var trace4 = { x: [1, 2, 3], y: [5, 5, 5], legendgroup: 'group2', line: {color: 'rgb(142, 124, 195)'}, mode: 'lines', name: 'second legend group - average', type: 'scatter' }; var data = [trace1, trace2, trace3, trace4]; var layout = { legend: { groupclick: 'toggleitem', groupdoubleclick: 'togglegroup' } }; Plotly.newPlot('myDiv', data, layout);
